2. 초보자를 위한 핵심 코드 원리 해설

이 게임이 어떻게 만들어졌는지 초보자의 시선에서 4가지 핵심 개념으로 나누어 설명해 드립니다.

① 캔버스(Canvas)와 그리기 도구

HTML의 <canvas> 태그는 그림을 그릴 수 있는 도화지 역할을 합니다. 자바스크립트에서 canvas.getContext("2d")를 호출하면 연필과 물감을 가진 붓(Context)을 얻게 되며, 이 붓을 이용해 네모(패들, 벽돌)나 동그라미(공)를 화면에 그려냅니다.

② 애니메이션 루프와 프레임 (`requestAnimationFrame`)

게임은 가만히 멈춰 있는 그림이 아니라 실시간으로 움직여야 합니다. requestAnimationFrame(draw) 함수는 화면을 아주 빠른 속도(초당 약 60회)로 지우고(clearRect), 공과 패들의 새로운 위치를 계산한 뒤 다시 그려주는 과정을 무한 반복하여 부드러운 움직임을 만들어냅니다.

③ 5단 컬러 벽돌과 점수 체계

벽돌은 2차원 배열(행과 열의 매트릭스) 구조로 생성됩니다. 각 행(Row)에 따라 고유한 색상과 점수를 부여했습니다.

  • 1단 (최상단): Cyan (50점)
  • 2단: Yellow (40점)
  • 3단: Blue (30점)
  • 4단: Green (20점)
  • 5단 (최하단): Red (10점)

④ 충돌 감지 (Collision Detection)

공이 이동하다가 벽돌이나 패들의 영역(네모 상자) 좌표 안에 들어오게 되면 이를 감지해냅니다. 벽돌과 부딪히면 공의 방향을 반대로 튕기고(dy = -dy), 해당 벽돌을 사라지게 한 뒤(status = 0) 점수를 누적합니다.

// 1. 캔버스 및 기본 변수 설정
const canvas = document.getElementById("gameCanvas");
const ctx = canvas.getContext("2d");

let score = 0;
let lives = 3;
let isPlaying = false; // 게임 진행 여부 플래그

// 공과 패들 좌표 설정
let x = canvas.width / 2;
let y = canvas.height - 30;
let dx = 3;
let dy = -3;
const ballRadius = 8;

const paddleHeight = 12;
const paddleWidth = 75;
let paddleX = (canvas.width - paddleWidth) / 2;

// 키보드 및 마우스 조작 이벤트
let rightPressed = false;
let leftPressed = false;

document.addEventListener("keydown", (e) => {
    if (e.key === "Right" || e.key === "ArrowRight" || e.key === "d") rightPressed = true;
    else if (e.key === "Left" || e.key === "ArrowLeft" || e.key === "a") leftPressed = true;
});
document.addEventListener("keyup", (e) => {
    if (e.key === "Right" || e.key === "ArrowRight" || e.key === "d") rightPressed = false;
    else if (e.key === "Left" || e.key === "ArrowLeft" || e.key === "a") leftPressed = false;
});
canvas.addEventListener("mousemove", (e) => {
    let relativeX = e.clientX - canvas.getBoundingClientRect().left;
    if (relativeX > 0 && relativeX < canvas.width) {
        paddleX = relativeX - paddleWidth / 2;
    }
});

// 5단 벽돌 설정
const brickRowCount = 5;
const brickColumnCount = 7;
const brickWidth = 58;
const brickHeight = 16;
const brickPadding = 8;
const brickOffsetTop = 25;
const brickOffsetLeft = 20;

const rowConfig = [
    { color: "#00f3ff", points: 50 },
    { color: "#ffff00", points: 40 },
    { color: "#0066ff", points: 30 },
    { color: "#00ff66", points: 20 },
    { color: "#ff2a6d", points: 10 }
];

let bricks = [];
function initBricks() {
    bricks = [];
    for (let c = 0; c < brickColumnCount; c++) {
        bricks[c] = [];
        for (let r = 0; r < brickRowCount; r++) {
            bricks[c][r] = { x: 0, y: 0, status: 1, color: rowConfig[r].color, points: rowConfig[r].points };
        }
    }
}
