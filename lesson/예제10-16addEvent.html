addEventListener() 원본 및 실용 코드 병합 분석 문서

본 문서는 표준 이벤트 등록 메서드인 `addEventListener()`를 활용한 기초 원본 코드와, 이를 실무적 인터랙션(데이터 갱신 및 시각적 피드백 결합)으로 발전시킨 실용 개선 코드를 함께 병합하여 비교·설명합니다.

1. 원본 소스코드 (기초 다중 리스너 테스트)

하나의 요소에 addEventListener를 이용해 여러 개의 이벤트 핸들러(알림창)를 중복 등록하는 원리를 보여줍니다.

<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>예제 10-6 addEventListener() 원본</title>
  <script>
    function addE() {
      var x = document.getElementById("txt");
      // 같은 요소에 클릭 이벤트를 두 개 등록한다
      x.addEventListener("click", add2);
      x.addEventListener("click", add3);
    }

    function add2() {
      alert("두번째 이벤트 호출");
    }
    function add3() {
      alert("세번째 이벤트 호출!");
    }
  </script>
<head>
<body>
  <h3>버튼을 클릭하세요</h3>
  <input id="txt" type="button" value="클릭" onclick="addE()">
</body>
</html>

2. 실용 개선 소스코드 (상태 갱신 및 시각적 피드백 결합)

원본 코드의 알림창 방식 대신, 하나의 버튼 클릭 이벤트에 '카운터 증가 로직'과 '버튼 애니메이션(스타일 변경) 로직'을 각각 독립된 함수로 나누어 다중 바인딩하는 실무형 패턴입니다.

<!DOCTYPE html>
<html>
<head>
  <meta charset="utf-8">
  <title>실용적인 addEventListener 활용 코드</title>
  <script>
    let clickCount = 0;

    // 기능 1: 데이터 상태 갱신 (카운터 증가)
    function updateCounter() {
        clickCount++;
        document.getElementById("counter").textContent = clickCount;
    }

    // 기능 2: UI 시각적 피드백 (버튼 배경색 및 텍스트 토글)
    function applyVisualFeedback() {
        const btn = document.getElementById("actionBtn");
        btn.style.backgroundColor = "#27ae60";
        setTimeout(() => {
            btn.style.backgroundColor = "#1769aa";
        }, 150);
    }

    // 페이지 로드 시 다중 이벤트 바인딩
    window.onload = function() {
        const btn = document.getElementById("actionBtn");
        btn.addEventListener("click", updateCounter);
        btn.addEventListener("click", applyVisualFeedback);
    };
  </script>
</head>
<body>
  <h3>실용형 다중 이벤트 리스너 실습</h3>
  <p>버튼 클릭 횟수: <b id="counter">0</b>회</p>
  <input id="actionBtn" type="button" value="통합 액션 버튼">
</body>
</html>

실시간 통합 실행 화면

[파트 1] 원본 예제 테스트

버튼을 처음 누르면 리스너가 등록되고, 이후 클릭부터 다중 알림창이 호출됩니다.

[파트 2] 실용 개선 예제 테스트

버튼 클릭 횟수: 0회

3. 핵심 비교 및 요약

구분 원본 코드 특징 실용 개선 코드 특징
목적 동일 요소에 여러 이벤트 리스너가 중복 등록될 수 있음을 증명 데이터 처리(상태)와 화면 시각 효과(UI)를 각각 분리하여 실무에 적용
사용자 경험(UX) 반복적인 알림창(alert)으로 인해 다소 번거로움 부드러운 카운터 갱신과 버튼 색상 애니메이션으로 직관적인 피드백 제공
결론 addEventListener는 전통적인 방식과 달리 하나의 DOM 요소에 여러 개의 독립된 함수를 충돌 없이 결합할 수 있는 강력한 표준 기능입니다.
